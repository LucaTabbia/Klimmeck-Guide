final RegExp _cloudinaryRelativePath = RegExp(r'^v\d+/\S+$');

/// URL del ritratto di un nemico dal suo `imagePath` (D-03): un URL `https`
/// resta com'è, un percorso Cloudinary relativo `v<n>/…` passa da
/// [CloudinaryAssets.url]; tutto il resto (i segnaposto `assets/images/…`
/// del DB, BE Phase 11 D-13) non si mostra finché l'utente non carica le
/// immagini (BE Phase 11 D-24).
String? enemyPortraitUrl(String? imagePath) {
  final path = imagePath?.trim();
  if (path == null || path.isEmpty) return null;
  final uri = Uri.tryParse(path);
  if (uri != null && uri.scheme == 'https' && uri.host.isNotEmpty) return path;
  if (_cloudinaryRelativePath.hasMatch(path)) return CloudinaryAssets.url(path);
  return null;
}
```

/// Ritratto di un nemico dal suo `imagePath` (D-03): SVG da [CachedSvg],
/// raster da [CachedNetworkImage]; percorso non risolvibile o download
/// fallito → nessun riquadro (mai un segnaposto nel bundle, mai uno spinner).
/// Unico widget per il foglio quest e il foglio esiti; la Phase 17 lo scurirà.
class EnemyPortrait extends StatelessWidget {
  const EnemyPortrait({super.key, required this.imagePath, this.height, this.cacheManager});

  static const String semanticLabel = 'Ritratto del nemico';

  final String? imagePath;
  final double? height;

  /// Solo per i test: la cache di default va in rete.
  final BaseCacheManager? cacheManager;

  /// Vero quando c'è qualcosa da caricare: chi impagina riserva lo spazio solo allora.
  static bool canShow(String? imagePath) => enemyPortraitUrl(imagePath) != null;

  @override
  Widget build(BuildContext context) {
    final url = enemyPortraitUrl(imagePath);
    if (url == null) return const SizedBox.shrink();
    return Semantics(label: semanticLabel, image: true, child: _image(url));
  }

  Widget _image(String url) {
    if (Uri.parse(url).path.toLowerCase().endsWith('.svg')) {
      return CachedSvg(url: url, height: height);
    }
    return CachedNetworkImage(
      imageUrl: url,
      height: height,
      fit: BoxFit.contain,
      cacheManager: cacheManager,
      placeholder: (_, _) => const SizedBox.shrink(),
      errorWidget: (_, _, _) => const SizedBox.shrink(),
    );
  }
}
```
