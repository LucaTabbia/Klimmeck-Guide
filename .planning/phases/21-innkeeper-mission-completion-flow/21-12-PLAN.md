// missionDraftCubit
sealed class MissionDraftState extends Equatable
final class MissionDraftLoading extends MissionDraftState {}
final class MissionDraftLoadFailed extends MissionDraftState { final InnkeeperFailure failure; }
final class MissionDraftAbsent extends MissionDraftState { final bool isStarting; final InnkeeperFailure? failure; final int failureSerial; }
final class MissionDraftReady extends MissionDraftState { final MissionDraft draft; final bool isSubmitting; final InnkeeperFailure? failure; final int failureSerial; MissionDraftReady copyWith(...); }
final class MissionDraftCompleted extends MissionDraftState { final List<QuestResult> results; }
final class MissionDraftDiscarded extends MissionDraftState {}
class MissionDraftCubit extends Cubit<MissionDraftState> with SafeEmit<MissionDraftState> {
  MissionDraftCubit({required InnkeeperRepository repository, required String pendingQuestId});
  Future<void> load();                                   // Loading → Absent | Ready | LoadFailed; also used as «Riprova»
  Future<void> start();                                  // Absent → Absent(isStarting) → Ready | Absent(failure, serial+1)
  Future<void> saveStep(MissionDraftStep step, {String? placeId});
  Future<void> saveFights(List<List<MissionEnemyInput>> enemiesPerFight, {MissionDraftStep? step}); // fights: fightsInputWithEnemies(draft, …)
  Future<void> simulate(int fightIndex);                 // enemies = served fight enemies → MissionEnemyInput.of
  Future<void> saveCorrections(int fightIndex, List<MissionCorrection> corrections, {MissionDraftStep? step}); // fights: fightsInputWithCorrections
  Future<void> savePrizes(List<MissionPrizeInput> prizes, {MissionDraftStep? step});
  Future<void> discard();                                // Ready → Discarded
  Future<void> complete();                               // Ready → Completed(results)
}
// Ready actions: ignored unless state is MissionDraftReady && !isSubmitting; emit copyWith(isSubmitting: true, failure: null) → Ready(newDraft) | Ready(old draft, failure, serial+1).

// missionParticipantsCubit
sealed class MissionParticipantsState; MissionParticipantsIdle; MissionParticipantsLoading; MissionParticipantsFailed { failure };
final class MissionParticipantsLoaded { final Map<String, Character> characters; final Map<String, PointOfInterest> places;
  String? get commonLocationId;                   // every character has the same status.location.id, else null
  String placeLabelFor(String? locationId);       // placeLabelOf(places[id]) or '' when unknown
  String? itemNameOf(String characterId, String itemId);   // from assets.ownedItems
  String? spellNameOf(String characterId, String spellId); // from status.spellStates }
class MissionParticipantsCubit { MissionParticipantsCubit({required InnkeeperRepository repository}); Future<void> load(List<String> characterIds); } // a second load while loading is ignored; reload allowed after Loaded/Failed

// missionEnemiesCubit
final class MissionEnemiesState { final List<MissionEnemy> enemies; final bool showAll; final String query; final bool isLoading; final InnkeeperFailure? failure;
  List<MissionEnemy> get visible; }   // query trimmed/lowercased, name contains; empty query = all, served order
class MissionEnemiesCubit { MissionEnemiesCubit({required InnkeeperRepository repository, required String pendingQuestId});
  Future<void> load(); Future<void> setShowAll(bool showAll); void setQuery(String query); }

// missionPrizeOptionsCubit
final class MissionPrizeOptionsState { final Map<String, MissionPrizeOptions> options; final Set<String> loading; final Map<String, InnkeeperFailure> failures; }
class MissionPrizeOptionsCubit { MissionPrizeOptionsCubit({required InnkeeperRepository repository, required String pendingQuestId});
  Future<void> loadFor(String characterId);        // no call when already loaded or loading; a failure can be retried
  Future<void> loadAll(List<String> characterIds); }
