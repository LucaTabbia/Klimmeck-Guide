/// Raggio in pixel di schermo entro cui il rilascio cade su un luogo interno (D-29).
const double teleportPlaceDropRadius = 32;
/// Regola di rilascio della sola pedina del teletrasporto (D-01, D-29): il luogo interno più vicino
/// entro [radius] pixel, altrimenti la città della regola del viaggio (Phase 7 D-01). Pura.
TeleportDestination? teleportDropTarget({
  required LatLng releasePoint,
  required List<TeleportDestination> places,
  required List<City> cities,
  required Offset Function(LatLng point) toScreen,
  double radius = teleportPlaceDropRadius,
});
// pawn_dragging.dart (mixin)
/// Regola di rilascio: la pedina del giocatore usa la città (Phase 7 D-01); la pedina del teletrasporto la sovrascrive (D-29).
void resolvePawnDrop(LatLng releasePoint) {
  final targetCity = MapUtils.dropTargetCity(releasePoint, dropCities);
  onPawnDragEnd(targetCity?.markerLocation?.location, targetCity);
}
// teleport_pawn_layer.dart
TeleportPawnLayer({required target, required places, required cities, required zoomNotifier, required mapController, required canDrag, required onDragUpdate, required void Function(TeleportDestination? destination) onDrop});
// teleport_poi_layer.dart
class TeleportPoiLayer extends StatelessWidget {
  const TeleportPoiLayer({super.key, required this.places, required this.zoomNotifier});
  static Key placeKeyFor(String poiId) => ValueKey('map.teleportPlace.$poiId');
  static const double dotSize = 14;
}
