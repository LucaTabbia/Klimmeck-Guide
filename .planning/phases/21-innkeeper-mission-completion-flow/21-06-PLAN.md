```dart
// lib/models/travel/teleport_destination.dart (Flutter-free: equatable + latlong2 only)
class TeleportDestination extends Equatable {
  const TeleportDestination({required this.poiId, required this.point, required this.label, required this.isCityMarker});
  final String poiId;
  final LatLng point;
  final String label;
  final bool isCityMarker;
  static TeleportDestination? ofCity(City? city);          // marker id + marker point; label = city.name ?? PoiType.city.label; isCityMarker true; null without marker/point
  static TeleportDestination? ofPlace(PointOfInterest poi);  // null when poi.location == null, when poi.type contains PoiType.city, or when placeLabelOf(poi) is empty
}
/// «<tipo> presso <città>» per un luogo interno, il nome della città per un marker (D-29).
String placeLabelOf(PointOfInterest poi);
/// Solo i luoghi interni con un punto, nell'ordine del BE.
List<TeleportDestination> innerPlacesOf(List<PointOfInterest> pointsOfInterest);
```
