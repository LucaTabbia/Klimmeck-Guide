   ```dart
   import 'package:dio/dio.dart';
   import 'package:flutter/foundation.dart';
   import 'package:klimmeck_guide/repository/services/auth/auth_token_service.dart';

   class AuthInterceptor extends Interceptor {
     AuthInterceptor(this._authTokenService);
     final AuthTokenService _authTokenService;

     @override
     Future<void> onRequest(
       RequestOptions options,
       RequestInterceptorHandler handler,
     ) async {
       try {
         final token = await _authTokenService.getAccessToken();
         if (token != null && token.isNotEmpty) {
           options.headers['Authorization'] = 'Bearer $token';
         }
       } catch (e) {
         if (kDebugMode) debugPrint('[AuthInterceptor] token fetch failed: $e');
       }
       handler.next(options);
     }
   }
   ```

   ```dart
   import 'package:gql_exec/gql_exec.dart';
   import 'package:gql_link/gql_link.dart';
   import 'package:graphql_flutter/graphql_flutter.dart';
   import 'package:klimmeck_guide/repository/services/auth/auth_token_service.dart';

   Link buildAuthLink(AuthTokenService authTokenService) {
     return Link.function((request, [forward]) async* {
       final token = await authTokenService.getAccessToken();
       final headers = <String, String>{
         if (token != null && token.isNotEmpty) 'Authorization': 'Bearer $token',
       };
       final updated = request.updateContextEntry<HttpLinkHeaders>(
         (prev) => HttpLinkHeaders(
           headers: {...?prev?.headers, ...headers},
         ),
       );
       yield* forward!(updated);
     });
   }
   ```

   ```dart
     // Phase 1: initialPayload letto una volta al setup.

         },

     );

     final httpWithAuth = authLink.concat(httpLink);
     final link = Link.split((r) => r.isSubscription, wsLink, httpWithAuth);
   ```
