```dart
/// Filtro comune a background e foreground (D-14): messaggio valido, utente noto, toggle attivo.
/// `null` = non mostrare nulla. Mai un'eccezione verso il chiamante.
Future<PushEvent?> acceptPushEvent(
  Map<String, Object?> data, {
  required String? userId,
  required NotificationPreferencesStore preferences,
}) async {
  final event = PushEvent.tryParse(data);
  if (event == null) {
    debugPrint('Push dropped: unknown or malformed message');
    return null;
  }
  if (userId == null) return null;
  try {
    return await preferences.isEnabled(userId, event.category) ? event : null;
  } on Object catch (error) {
    debugPrint('Push dropped: preferences unavailable (${error.runtimeType})');
    return null;
  }
}

/// Decisione dell'isolate in background (D-12): una notifica di sistema se [acceptPushEvent] la ammette.
Future<void> handleDataMessage(
  Map<String, Object?> data, {
  required ActiveUserStore activeUser,
  required NotificationPreferencesStore preferences,
  required LocalNotificationsPort notifier,
}) async {
  final event = await acceptPushEvent(data, userId: await _activeUserId(activeUser), preferences: preferences);
  if (event == null) return;
  await notifier.show(event);
}

Future<String?> _activeUserId(ActiveUserStore store) async {
  try {
    return await store.read();
  } on Object catch (error) {
    debugPrint('Push dropped: active user unavailable (${error.runtimeType})');
    return null;
  }
}
```

   ```dart
   class FlutterLocalNotificationsAdapter implements LocalNotificationsPort {
     FlutterLocalNotificationsAdapter([FlutterLocalNotificationsPlugin? plugin]) : _plugin = plugin ?? FlutterLocalNotificationsPlugin();

     static const InitializationSettings _settings = InitializationSettings(android: AndroidInitializationSettings(PushChannel.smallIcon));
     static const AndroidNotificationChannel _channel = AndroidNotificationChannel(PushChannel.id, PushChannel.name, description: PushChannel.description, importance: Importance.high);
     static const NotificationDetails _details = NotificationDetails(
       android: AndroidNotificationDetails(PushChannel.id, PushChannel.name, channelDescription: PushChannel.description, importance: Importance.high, priority: Priority.high),
     );

     final FlutterLocalNotificationsPlugin _plugin;
     final StreamController<NotificationIntent> _taps = StreamController<NotificationIntent>.broadcast();

     /// Isolate principale: canale + tap. Creare il canale non mostra il prompt (target 36).
     @override
     Future<void> initialize() async {
       await _plugin.initialize(_settings, onDidReceiveNotificationResponse: _onTap);
       await _createChannel();
     }

     /// Isolate in background: solo canale, nessun callback (i tap arrivano all'isolate principale).
     Future<void> initializeWithoutTaps() async {
       await _plugin.initialize(_settings);
       await _createChannel();
     }

     @override
     Future<void> show(PushEvent event) => _plugin.show(event.notificationId, event.title, event.body, _details, payload: event.intent.toPayload());

     @override
     Stream<NotificationIntent> get taps => _taps.stream;

     @override
     Future<NotificationIntent?> launchIntent() async {
       final details = await _plugin.getNotificationAppLaunchDetails();
       if (details == null || !details.didNotificationLaunchApp) return null;
       return NotificationIntent.tryFromPayload(details.notificationResponse?.payload);
     }

     @override
     Future<void> cancelAll() => _plugin.cancelAll();

     Future<void> _createChannel() async => _plugin
         .resolvePlatformSpecificImplementation<AndroidFlutterLocalNotificationsPlugin>()
         ?.createNotificationChannel(_channel);

     void _onTap(NotificationResponse response) {
       final intent = NotificationIntent.tryFromPayload(response.payload);
       if (intent != null && !_taps.isClosed) _taps.add(intent);
     }
   }
   ```

   ```dart
   /// Limite accettato (D-10): su Android 7-12 un'app disattivata a mano risulta notDetermined.
   class PermissionHandlerNotificationPermission implements NotificationPermissionPort {
     const PermissionHandlerNotificationPermission();
     @override
     Future<NotificationPermissionStatus> status() async {
       final status = await Permission.notification.status;
       if (status.isGranted) return NotificationPermissionStatus.granted;
       if (status.isPermanentlyDenied) return NotificationPermissionStatus.denied;
       return await Permission.notification.shouldShowRequestRationale
   ```
