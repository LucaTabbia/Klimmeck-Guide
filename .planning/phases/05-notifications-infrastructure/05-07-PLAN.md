   ```dart
   /// Tap in attesa di essere consumato da MainScreen (D-15). Vive sopra AuthGate: sopravvive a
   /// splash e sign-in (tap da killed), si svuota a fine sessione e sulla pagina di creazione.
   class NotificationIntentCubit extends Cubit<NotificationIntentState> implements NotificationIntentSink {
     NotificationIntentCubit(Stream<AuthState> authStates) : super(const NotificationIntentState()) {
       _authSubscription = authStates.listen(_onAuthState);
     }

     late final StreamSubscription<AuthState> _authSubscription;
     AuthState? _lastAuthState;

     @override
     void offer(NotificationIntent intent) {
       if (_isOnCreationPage) return;
       emit(NotificationIntentState(pending: intent));
     }

     /// Restituisce e cancella il tap in attesa.
     NotificationIntent? consume() {
       final pending = state.pending;
       if (pending != null) emit(const NotificationIntentState());
       return pending;
     }

     void _onAuthState(AuthState next) {
       final previous = _lastAuthState;
       _lastAuthState = next;
       final endedSession = next is AuthUnauthenticated && previous is AuthAuthenticated;
       if (endedSession || _isOnCreationPage) _clear();
     }

     bool get _isOnCreationPage => switch (_lastAuthState) {
       AuthAuthenticated(:final user) => user.currentCharacter == null,
       _ => false,
     };

     void _clear() {
       if (state.pending != null && !isClosed) emit(const NotificationIntentState());
     }

     @override
     Future<void> close() async {
       await _authSubscription.cancel();
       return super.close();
     }
   }
   ```

   ```dart
   /// Consuma il tap in attesa (D-15): chiude fogli e dialog sul Navigator radice, salta al tab
   /// Mappa e apre l'eventuale popup del tipo. Montato solo dove esiste il PageView (Pitfall 14).
   class NotificationIntentLanding extends StatefulWidget {
     const NotificationIntentLanding({super.key, required this.pageController, required this.child, this.popups = defaultNotificationPopups});

     /// Indice di WorldMap in `MainScreen._buildPages`.
     static const int worldMapPageIndex = 0;

     final PageController pageController;
     final Widget child;
     final Map<NotificationCategory, NotificationPopup> popups;
     …

   class _NotificationIntentLandingState extends State<NotificationIntentLanding> {
     @override
     void initState() {
       super.initState();
       WidgetsBinding.instance.addPostFrameCallback((_) => _consumePendingIntent());
     }

     @override
     Widget build(BuildContext context) => BlocListener<NotificationIntentCubit, NotificationIntentState>(
       listenWhen: (previous, current) => current.pending != null && previous.pending != current.pending,
       listener: (_, _) => _consumePendingIntent(),

       widget.popups[intent.category]?.call(context, intent);
   ```
