abstract interface class LegalConsentStore {
  Future<String?> readAcceptedVersion();
  Future<void> remember(String version);
  Future<void> forget();
}
class SharedPreferencesLegalConsentStore implements LegalConsentStore {
  SharedPreferencesLegalConsentStore({SharedPreferencesAsync Function()? preferences});
  static const String key = 'klimmeck.legal.acceptedVersion';
}

// lib/screens/signIn/consentCubit/legal_consent_cubit.dart (+ part legal_consent_state.dart)
class LegalConsentState extends Equatable { const LegalConsentState({this.isAccepted = false}); final bool isAccepted; }
class LegalConsentCubit extends Cubit<LegalConsentState> with SafeEmit<LegalConsentState> {

  Future<void> withdraw();  // emits unticked, then forgets
