```dart
// lib/repository/services/sync/sync_health.dart
enum SyncSource { socket, userStream, characterStream }

abstract interface class SyncHealth {
  /// Sorgente viva: connessione stabilita, evento o snapshot ricevuto.
  void markHealthy(SyncSource source);
  /// Sorgente in recupero (socket giù, stream fallito/chiuso in re-iscrizione).
  void markDegraded(SyncSource source);
  /// Stream rifiutato dal backend (FORBIDDEN/NOT_FOUND): nessun retry.
  void markTerminated(SyncSource source);
  /// Il proprietario della sorgente è stato chiuso (shell ri-chiavata, logout).
  void release(SyncSource source);
  /// Un refetch HTTP è riuscito: prova di vita, riapre la finestra se degradato.
  void touch();
}

// lib/repository/services/sync/sync_watchdog.dart
class SyncWatchdog implements SyncHealth {
  SyncWatchdog({required Future<void> Function() onExpired, Duration window = defaultWindow});
  static const Duration defaultWindow = Duration(minutes: 3);
  bool get isDegraded;
  void suspend();   // app in background
  void resume();    // app in foreground: fresh window if still degraded
  void dispose();
}

// lib/repository/services/sync/app_lifecycle_events.dart
abstract interface class AppLifecycleEvents {
  Stream<void> get backgrounded;
  Stream<void> get foregrounded;
}
class AppLifecycleEventBus implements AppLifecycleEvents {
  void notifyBackgrounded();
  void notifyForegrounded();
  Future<void> close();
}

// lib/repository/services/sync/user_refresh_requests.dart
abstract interface class UserRefreshRequests {
  void requestUserRefresh();
}
class UserRefreshRequestBus implements UserRefreshRequests {
  Stream<void> get requests;
  Future<void> close();
}

// lib/utils/silent_retry.dart
Future<T?> silentRetry<T>(
  Future<T> Function() attempt, {
  required bool Function() isCancelled,
  bool Function(Object error) shouldRetry = _alwaysRetry,
  void Function(Object error)? onGiveUp,
  int maxAttempts = defaultSilentRetryAttempts,   // const int defaultSilentRetryAttempts = 8;
  Duration maxDelay = const Duration(seconds: 60),
});
```

```dart
const int defaultSilentRetryAttempts = 8;

bool _alwaysRetry(Object error) => true;

Future<T?> silentRetry<T>(
  Future<T> Function() attempt, {
  required bool Function() isCancelled,
  bool Function(Object error) shouldRetry = _alwaysRetry,
  void Function(Object error)? onGiveUp,
  int maxAttempts = defaultSilentRetryAttempts,
  Duration maxDelay = const Duration(seconds: 60),
}) async {
  for (var n = 0; n < maxAttempts; n++) {
    if (isCancelled()) return null;
    try {
      return await attempt();
    } catch (error) {
      final isLastAttempt = n == maxAttempts - 1;
      if (!shouldRetry(error) || isLastAttempt) {
        if (!isCancelled()) onGiveUp?.call(error);
        return null;
      }
      await Future<void>.delayed(exponentialBackoff(n, max: maxDelay));
    }
  }
  return null;
}
```
Dartdoc (Italian): transient failures stay silent (D-14); the cap of 8 attempts with 1→60 s backoff spans ≈ 2 minutes, inside the 3-minute watchdog window, so a sync that never recovers is ended by the watchdog, not by an endless loop (graphql.md anti-pattern).
