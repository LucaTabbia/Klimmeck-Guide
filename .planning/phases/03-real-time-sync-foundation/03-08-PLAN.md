/// `AuthGate`, quindi logout, revoca, sync persa e cambio account lo smontano e

/// consumer di `userUpdated` e osserva il socket una sola volta per sessione.
class SessionSyncScope extends StatelessWidget {
  ...
  @override
  Widget build(BuildContext context) {
    return MultiRepositoryProvider(
      providers: [
        RepositoryProvider<SyncWatchdog>(
          create: (_) => SyncWatchdog(onExpired: () => terminator.terminateSession(UnauthenticatedReason.syncLost)),
          dispose: (watchdog) => watchdog.dispose(),
        ),
        RepositoryProvider<SyncHealth>(create: (context) => context.read<SyncWatchdog>()),
        RepositoryProvider<UserRefreshRequestBus>(create: (_) => UserRefreshRequestBus(), dispose: (bus) => bus.close()),

          lifecycle: context.read<AppLifecycleEvents>(),

        child: child,

}

class AppLifecycleBridge extends StatefulWidget {
  const AppLifecycleBridge({super.key, required this.bus, required this.child});

  void initState() { super.initState(); _listener; }

}
