```dart
class AmbushReportCarousel extends StatefulWidget {
  static const Key pagesKey = ValueKey('ambushReportCarousel.pages');
  /// D-19: with several reports each page takes this fraction of the viewport so the neighbours peek in.
  static const double peekViewportFraction = 0.86;
  /// D-19: one nudge at opening (forward, then back on page 1), only with several reports.
  static const double nudgeDistance = 28;
  static const Duration nudgeOut = Duration(milliseconds: 240);
  static const Duration nudgeBack = Duration(milliseconds: 280);
  ...
}
class _AmbushReportCarouselState extends State<AmbushReportCarousel> {
  late final PageController _controller = PageController(
    viewportFraction: widget.results.length > 1 ? AmbushReportCarousel.peekViewportFraction : 1.0,
  );
  @override void initState() { super.initState(); if (widget.results.length > 1) WidgetsBinding.instance.addPostFrameCallback((_) => unawaited(_nudge())); }
  Future<void> _nudge() async {
    if (!mounted || !_controller.hasClients) return;
    await _controller.animateTo(_controller.offset + AmbushReportCarousel.nudgeDistance, duration: AmbushReportCarousel.nudgeOut, curve: Curves.easeOut);
    if (!mounted || !_controller.hasClients) return;
    await _controller.animateTo(0, duration: AmbushReportCarousel.nudgeBack, curve: Curves.easeIn);
  }
  @override void dispose() { _controller.dispose(); super.dispose(); }
  @override Widget build(BuildContext context) => PageView(key: AmbushReportCarousel.pagesKey, controller: _controller, children: [...unchanged...]);
}
```
