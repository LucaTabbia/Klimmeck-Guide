/// OperationException alla UI). Il ruolo è verificato dal BE (`@Roles(innkeeper)`,
/// ADMIN-07): qui `FORBIDDEN` è solo un esito da mostrare.
class InnkeeperRepository {
  InnkeeperRepository({required KlimmeckGraphQl graphQl, this.timeout = defaultTimeout}) : _graphQl = graphQl;
  static const Duration defaultTimeout = Duration(seconds: 15);
  final KlimmeckGraphQl _graphQl; final Duration timeout;
  Stream<List<PendingRequest>> watchQueue() => _graphQl.subscribeToPendingRequests();
  Future<List<PendingRequest>> fetchQueue() => _guard(_graphQl.getPendingRequests);
  /// Lettura del GM di un personaggio qualsiasi (BE Phase 3): form e teletrasporto.
  Future<Character> readCharacter(String characterId) => _guard(() => _graphQl.getCharacter(characterId));
  Future<List<QuestResult>> complete(CompletePendingRequestInput input) => _guard(() => _graphQl.completePendingRequest(input));
  Future<void> teleport({required String characterId, required String poiId}) async {
    final teleported = await _guard(() => _graphQl.teleportCharacter(TravelRequest(id: characterId, poiId: poiId)));
    if (!teleported) throw const InnkeeperException(InnkeeperFailure.declined);
  }
  Future<T> _guard<T>(Future<T> Function() call) async {
    try { return await call().timeout(timeout); }

    on GraphQlAccessDeniedException catch (error) { throw InnkeeperException(error.code == forbiddenCode ? InnkeeperFailure.forbidden : InnkeeperFailure.notFound); }
    on TimeoutException { throw const InnkeeperException(InnkeeperFailure.timedOut); }
    on FormatException { throw const InnkeeperException(InnkeeperFailure.malformed); }
    on Object { throw const InnkeeperException(InnkeeperFailure.unknown); }
  }
}
