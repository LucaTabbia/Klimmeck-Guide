   /// Modello puro dello strappo (D-06): numeri provvisori, rivisti dall'utente (D-19).
   class TearGesture {
     const TearGesture({this.travelFraction = 0.45, this.flingProgress = 0.35, this.flingVelocity = -900, this.blockedCap = 0.12});
     /// Frazione della larghezza da trascinare per lo strappo pieno.
     final double travelFraction;
     final double flingProgress;
     /// Velocità orizzontale (px/s, negativa = sinistra) oltre cui un lancio strappa.
     final double flingVelocity;
     /// Massimo avanzamento di un foglio che resiste (swipe bloccato).
     final double blockedCap;

     double progress(double dragDx, double width) => width <= 0 ? 0 : (-dragDx / (width * travelFraction)).clamp(0.0, 1.0);
     bool shouldTear(double progress, double velocityX) => progress >= 1 || (progress >= flingProgress && velocityX <= flingVelocity);
     double resisted(double progress) => progress < blockedCap ? progress : blockedCap;
   }

   /// Scatta una sola volta per trascinamento (haptic della soglia).
   class ThresholdLatch {
     bool _hasFired = false;
     bool cross(bool isReached) {
       if (!isReached || _hasFired) return false;
       _hasFired = true;
       return true;
     }
     void reset() => _hasFired = false;
   }
   ```

   enum TearSide { left, right }

   /// Taglio verticale frastagliato a metà foglio, deterministico (mai Random):
   /// le due metà condividono lo stesso bordo.
   class TearClipper extends CustomClipper<Path> {
     const TearClipper({required this.side});
     final TearSide side;
     static const int teeth = 14;
     static const double toothDepth = 0.04;

     static List<Offset> edgePoints(Size size) => [
       for (var i = 0; i <= teeth; i++)
         Offset(size.width * (0.5 + (i.isEven ? -toothDepth : toothDepth)), size.height * i / teeth),
     ];

     @override
     Path getClip(Size size) {
       final edge = edgePoints(size);
       final outerX = side == TearSide.left ? 0.0 : size.width;
       final path = Path()..moveTo(outerX, 0);
       for (final point in edge) {
         path.lineTo(point.dx, point.dy);
       }
       return path
         ..lineTo(outerX, size.height)
         ..close();
     }

     @override
     bool shouldReclip(TearClipper oldClipper) => oldClipper.side != side;
   }
   ```
