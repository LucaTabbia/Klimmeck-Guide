```dart
KGNotification? spellNoticeFor(SpellOutcome outcome) {
  final description = switch (outcome.kind) {
    SpellOutcomeKind.blockedSlotsFull => 'Slot pieni: disequipaggia una magia',
    SpellOutcomeKind.blockedNoSlots => 'Nessuno slot disponibile',
    SpellOutcomeKind.failed => _failureLine(outcome.failure ?? SpellEquipFailure.unknown),

  if (description == null) return null;
  return KGNotification(title: outcome.operation == SpellOperation.unequip ? spellUnequipNoticeTitle : spellEquipNoticeTitle, description: description);
}
```

```dart
/// Esiti delle magie a livello di sessione (D-16, D-17): l'overlay segue ogni
/// `characterUpdated`; banner e riallineamento sopravvivono al cambio tab (F3).
class SpellOutcomeListener extends StatelessWidget {
  const SpellOutcomeListener({super.key, required this.child});
  final Widget child;
  /// Dopo questi rifiuti la vista è stantia e il BE non emette nulla: rilettura silenziosa (F1).
  static const Set<SpellEquipFailure> _staleViewFailures = {
    SpellEquipFailure.alreadyActive, SpellEquipFailure.notActive, SpellEquipFailure.notKnown,
    SpellEquipFailure.slotsFull, SpellEquipFailure.notFound,
  };
  @override
  Widget build(BuildContext context) => MultiBlocListener(listeners: [
    BlocListener<CharacterCubit, CharacterState>(
      listenWhen: (previous, current) => current is CharacterLoaded && (previous is! CharacterLoaded || previous.character != current.character),
      listener: (context, state) => context.read<SpellsCubit>().sync((state as CharacterLoaded).character),
    ),
    BlocListener<SpellsCubit, SpellsState>(
      listenWhen: (previous, current) => current.outcome != null && previous.outcome != current.outcome,
      listener: (context, state) => _onOutcome(context, state.outcome!),
    ),
  ], child: child);
  …_onOutcome: notice → publishNotice(notice, dedupKey: 'spell|${outcome.kind.name}|${outcome.failure?.name ?? '-'}|${outcome.spellId}'); if kind == failed && _staleViewFailures.contains(failure) → context.read<CharacterCubit>().refreshCharacter();
}
```
