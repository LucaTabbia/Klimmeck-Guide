/// Tap su una magia (D-06..D-08, D-11): l'intento lo decide `SpellsCubit` sulla
/// vista ottimistica; qui solo l'aptica.
SpellToggleIntent toggleSpellWithFeedback(BuildContext context, {required Character character, required String spellId}) {
  final intent = context.read<SpellsCubit>().toggle(character: character, spellId: spellId);
  switch (intent) {
    case SpellToggleIntent.equip || SpellToggleIntent.unequip: HapticFeedback.lightImpact();
    case SpellToggleIntent.blockedSlotsFull || SpellToggleIntent.blockedNoSlots: HapticFeedback.selectionClick();
    case SpellToggleIntent.ignoredPending || SpellToggleIntent.ignoredUnknown: break;
  }
  return intent;
}
