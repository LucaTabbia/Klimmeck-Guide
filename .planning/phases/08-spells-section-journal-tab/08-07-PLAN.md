```dart
/// Countdown «prossimo tra X» sull'ora del server (D-04): ticchetta solo
/// finché almeno una magia mostra un countdown. Creato e chiuso dalla sezione
/// Magie: vive solo mentre la pagina del Diario è a schermo (D-16).
class SpellRecoveryCubit extends Cubit<SpellRecoveryState> with SafeEmit<SpellRecoveryState> {
  SpellRecoveryCubit({required ServerClockService clock, Duration tick = defaultTick}) : _clock = clock, _tick = tick, super(const SpellRecoveryState({})) {
    _offsetSubscription = clock.offsetChanges.listen((_) => _refresh());
  }
  static const Duration defaultTick = Duration(seconds: 1);
  static List<SpellRecoveryInput> inputsOf(Spellbook spellbook) => [ … per entry … ];
  final ServerClockService _clock; final Duration _tick;
  late final StreamSubscription<void> _offsetSubscription;
  Timer? _timer;
  List<SpellRecoveryInput> _inputs = const [];
  void follow(List<SpellRecoveryInput> inputs) { _inputs = List.unmodifiable(inputs); _refresh(); }
  void _refresh() {
    final now = _clock.now();
    final statuses = {for (final input in _inputs) input.spellId: spellRecoveryStatusOf(input, now)};
    emit(SpellRecoveryState(Map.unmodifiable(statuses)));
    if (statuses.values.any((status) => status is SpellRecoveryCountdown)) { _timer ??= Timer.periodic(_tick, (_) => _refresh()); } else { _stopTimer(); }
  }
  void _stopTimer() { _timer?.cancel(); _timer = null; }
  @override Future<void> close() { _stopTimer(); unawaited(_offsetSubscription.cancel()); return super.close(); }
}
```
