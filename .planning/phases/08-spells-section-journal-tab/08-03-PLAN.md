/// Segnalibro rosso degli oggetti equipaggiati e delle magie attive (D-14):

/// visibile; non segnato: sporge di [peek]. Con [onTap] fa da interruttore.
class BookmarkRibbon extends StatelessWidget {
  const BookmarkRibbon({super.key, required this.isMarked, required this.length, required this.peek, this.width, this.left = 5, this.curve = Curves.easeInOut, this.onTap});
  static const Duration slideDuration = Duration(milliseconds: 300);
  static final String _url = CloudinaryAssets.url(CloudinaryAssets.bookmarkRed);
  /// Altezza dello SVG prima della rotazione; senza [width] lo SVG tiene le sue proporzioni naturali.
  final double length; final double? width;
  final bool isMarked; final double peek; final double left; final Curve curve; final VoidCallback? onTap;
  @override
  Widget build(BuildContext context) {
    final svg = width == null
        ? CachedSvg(url: _url, height: length)                                   // today's sheet tree
        : SizedBox(height: length, width: width, child: CachedSvg(url: _url));  // today's card tree
    final ribbon = RotatedBox(quarterTurns: 3, child: svg);
    return AnimatedPositioned(
      duration: slideDuration, curve: curve,
      top: isMarked ? 0 : -length + peek, left: left,
      child: onTap == null ? ribbon : GestureDetector(onTap: onTap, child: ribbon),
    );
  }
}
