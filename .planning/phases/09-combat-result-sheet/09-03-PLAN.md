class QuestResultRepository {
  QuestResultRepository({required KlimmeckGraphQl graphQl, this.timeout = defaultTimeout}) : _graphQl = graphQl;
  static const Duration defaultTimeout = Duration(seconds: 15);
  final KlimmeckGraphQl _graphQl; final Duration timeout;
  Stream<QuestResult> watch(String characterId) => _graphQl.subscribeToQuestResults(characterId);
  Future<QuestResult> fetch(String resultId) => _guard(() => _graphQl.getQuestResult(resultId));
  Future<T> _guard<T>(Future<T> Function() call) async {
    try { return await call().timeout(timeout); }
    on OperationException catch (error) { throw QuestResultException(questResultFailureFrom(error)); }
    on TimeoutException { throw const QuestResultException(QuestResultFailure.timedOut); }
    on FormatException { throw const QuestResultException(QuestResultFailure.malformed); }
    on Object { throw const QuestResultException(QuestResultFailure.unknown); }
  }
}
