enum QuestType {
  …
  study(imageId: ''),
  arcane(imageId: ''),
  shamanic(imageId: ''),
  theVoid(imageId: '', wireName: 'void'),
  heal(imageId: ''),
  …;

  /// Nome con cui il backend serviva il tipo `arcane` prima della BE Phase 24
  /// (D-20): solo letto, mai scritto.
  static const String _legacyArcaneWireName = 'magic';

  String get wireName => _wireName ?? name;

  static QuestType? fromWire(Object? value) {
    if (value == _legacyArcaneWireName) return arcane;
    for (final type in values) {
      if (type.wireName == value) return type;
    }
    return null;
  }

  const QuestType({required String imageId, String? wireName})
    : _imageId = imageId,
      _wireName = wireName;
}
