// companion_vitality_state.dart (part of the cubit)
class CompanionVitalityState extends Equatable {
  const CompanionVitalityState({this.vitalities = const {}, this.potionWaits = const {}});
  final Map<String, CompanionVitality> vitalities; // by catalogue pet id
  final Map<String, Duration> potionWaits;         // only companions still waiting
  CompanionVitality? vitalityOf(String petId) => vitalities[petId];
  Duration? potionWaitOf(String petId) => potionWaits[petId];
}
// companion_vitality_cubit.dart
class CompanionVitalityCubit extends Cubit<CompanionVitalityState> with SafeEmit<CompanionVitalityState> {
  CompanionVitalityCubit({required ServerClockService clock, Duration tick = defaultTick});
  static const Duration defaultTick = Duration(seconds: 1);
  void follow(List<OwnedPet> owned, {String? equippedPetId});
}

class CompanionVitalityScope extends StatelessWidget { const CompanionVitalityScope({super.key, required this.child}); }
