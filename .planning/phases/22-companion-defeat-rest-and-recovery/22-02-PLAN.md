sealed class CompanionVitality extends Equatable {
  const CompanionVitality({required this.life, required this.maxLifePoints});
  final int life;
  final int? maxLifePoints;
}
final class CompanionSteady extends CompanionVitality { const CompanionSteady({required super.life, required super.maxLifePoints}); }
final class CompanionResting extends CompanionVitality { const CompanionResting({required super.life, required super.maxLifePoints, required this.remaining}); final Duration remaining; }
final class CompanionDefeated extends CompanionVitality { const CompanionDefeated({required super.life, required super.maxLifePoints, required this.remaining}); final Duration remaining; }
CompanionVitality companionVitalityOf(OwnedPet owned, DateTime now, {required bool isEquipped});
Duration? companionPotionWaitOf(OwnedPet owned, DateTime now);

// lib/models/companion/companion_display_name.dart
String companionDisplayName({required String? name, required Pet pet}); // trimmed non-empty name, else pet.name ?? ''
String companionDisplayNameOf(OwnedPet owned) => companionDisplayName(name: owned.name, pet: owned.pet);

// lib/utils/name_rules.dart
enum NameError { required, tooShort, tooLong, invalidCharacters }
const int nameMinLength = 2;
const int nameMaxLength = 20;
String normalizedNameOf(String raw);
NameError? nameErrorOf(String raw);
extension NameErrorMessage on NameError { String? get message; }

// lib/utils/companion_lines.dart (copy VERBATIM)
String? companionLifeLine(int life, int? maxLifePoints);           // 'Vita: $life / $maxLifePoints', null when max null or <= 0
String companionRestingLine(Duration remaining);                   // 'A riposo: in forze fra ${formatGameDuration(remaining)}'
String companionDefeatedLine(Duration remaining);                  // 'Sconfitto: torna fra ${formatGameDuration(remaining)}'
String? companionStatusLine(CompanionVitality vitality);           // Resting → resting line, Defeated → defeated line, Steady → null
String companionPotionWaitLine(Duration remaining);                // 'Pozione: ${potionReadyInLine(remaining)}'
String? companionRecoveryTimeLine(Duration? recovery);             // 'Recupero completo: ${formatGameDuration(recovery)}', null without
String giveToCompanionLabel(String name);                          // 'Dai a $name'
String companionNamePromptTitle(String species);                   // 'Come chiamerai il tuo $species?'
const String companionNameFieldLabel = 'Nome';
const String companionNameConfirmLabel = 'Chiamalo così';
const String companionNameSkipLabel = 'Più tardi';
String renameCompanionLabel({required bool hasName});              // hasName ? 'Cambia nome' : 'Dai un nome'
