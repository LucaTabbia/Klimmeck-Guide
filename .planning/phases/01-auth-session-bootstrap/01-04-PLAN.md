// lib/screens/auth/cubit/auth_state.dart
sealed class AuthCubitState extends Equatable { const AuthCubitState(); }
class AuthInitializing extends AuthCubitState { const AuthInitializing(); @override List<Object?> get props => []; }
class AuthenticatedShell extends AuthCubitState {
  final String userId;
  const AuthenticatedShell(this.userId);
  @override List<Object?> get props => [userId];
}
class AuthLoggingOut extends AuthCubitState { const AuthLoggingOut(); @override List<Object?> get props => []; }
class AuthUnauthenticatedShell extends AuthCubitState {
  final UnauthReason reason;   // expose reason so SignInScreen can show "session expired" message
  const AuthUnauthenticatedShell(this.reason);
  @override List<Object?> get props => [reason];
}

// lib/screens/auth/cubit/auth_cubit.dart
class AuthCubit extends Cubit<AuthCubitState> {
  AuthCubit({required AuthTokenService authService}) : super(const AuthInitializing());

  /// Triggers bootstrap of the underlying service; called once at app start.
  Future<void> bootstrap();

  /// Shows the confirmation dialog; on confirm, calls authService.logout().
  Future<void> logout({required BuildContext context});

  /// Combined logout + login with force_verify (D-14).
  Future<void> switchAccount({required BuildContext context});

  @override Future<void> close(); // cancel stream subscription
}

// lib/screens/signIn/cubit/sign_in_state.dart
sealed class SignInState extends Equatable { const SignInState(); }
class SignInIdle extends SignInState { const SignInIdle(); @override List<Object?> get props => []; }
class SignInInProgress extends SignInState { const SignInInProgress(); @override List<Object?> get props => []; }
class SignInError extends SignInState {
  final String message; // e.g. "Errore di connessione, riprova"
  const SignInError(this.message);
  @override List<Object?> get props => [message];
}

// lib/screens/signIn/cubit/sign_in_cubit.dart
class SignInCubit extends Cubit<SignInState> {
  SignInCubit({required AuthTokenService authService}) : super(const SignInIdle());
  Future<void> login();
}

// lib/shared/components/modal/logout_confirmation_dialog.dart
class LogoutConfirmationDialog extends StatelessWidget { const LogoutConfirmationDialog({super.key}); }
/// Returns true if user confirmed, false/null if cancelled or dismissed.
Future<bool?> showLogoutConfirmationDialog(BuildContext context);
