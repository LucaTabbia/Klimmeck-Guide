// lib/config/insecure_transport_error.dart
/// URL del backend in chiaro in una build release (D-03b): porta solo i NOMI
/// delle chiavi, mai gli URL.
class InsecureTransportError extends Error {
  InsecureTransportError(List<String> offendingKeys)
    : offendingKeys = List.unmodifiable(offendingKeys);

  final List<String> offendingKeys;

  @override
  String toString() => 'InsecureTransportError(${offendingKeys.join(', ')})';
}

// lib/config/env_config.dart (new members)
static void requireSecureTransport({bool isRelease = kReleaseMode}) {
  final offending = _cleartextKeys();
  if (offending.isEmpty) return;
  if (isRelease) throw InsecureTransportError(offending);
  appLog('[EnvConfig] cleartext backend URLs outside release: ${offending.join(', ')}');
}

static List<String> _cleartextKeys() => [
  if (!_hasScheme(baseUrl, 'https')) 'BASE_URL',
  if (!_hasScheme(graphqlHttpUrl, 'https')) 'GRAPHQL_HTTP_URL',
  if (!_hasScheme(graphqlWsUrl, 'wss')) 'GRAPHQL_WS_URL',
];

static bool _hasScheme(String url, String scheme) =>
    Uri.tryParse(url)?.scheme.toLowerCase() == scheme;

// lib/bootstrap/app_bootstrap.dart (new)
/// Schermata che sostituisce l'app in release con URL in chiaro (D-03b), o null.
@visibleForTesting
Widget? secureTransportBlocker({required bool isRelease}) {
  try {
    EnvConfig.requireSecureTransport(isRelease: isRelease);
    return null;
  } on InsecureTransportError {
    return const InsecureTransportScreen();
  }
}

Future<void> bootstrapApp({
  required AuthCompositionFactory buildAuth,
  bool isRelease = kReleaseMode,
}) async {
  WidgetsFlutterBinding.ensureInitialized();
  final blocker = secureTransportBlocker(isRelease: isRelease);
  if (blocker != null) return runApp(blocker);
  // … unchanged: bootstrapPush, auth, holder, rest, SystemChrome, runApp
}
