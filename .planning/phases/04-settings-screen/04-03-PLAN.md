   ```dart
   /// Preferenze notifiche per utente, solo sul device (D-10/D-11). Non è un
   /// artefatto di sessione: logout e teardown non la toccano. Leggibile anche
   /// fuori dall'albero widget (isolate in background di Phase 5, D-14).
   abstract interface class NotificationPreferencesStore {
     Future<bool> isEnabled(String userId, NotificationCategory category);
     Future<Map<NotificationCategory, bool>> readAll(String userId);
     Future<void> setEnabled(String userId, NotificationCategory category, {required bool enabled});
   }

   /// `SharedPreferencesAsync` legge sempre lo storage di piattaforma: nessuna
   /// cache per isolate da invalidare. Istanza creata alla prima lettura
   /// (il costruttore lancia senza plugin registrato, es. nei test).
   class SharedPreferencesNotificationPreferencesStore implements NotificationPreferencesStore {
     SharedPreferencesNotificationPreferencesStore({SharedPreferencesAsync Function()? preferences})
       : _createPreferences = preferences ?? (() => SharedPreferencesAsync());
     static const String keyPrefix = 'klimmeck.notifications';
     static const bool defaultEnabled = true;
     static String keyFor(String userId, NotificationCategory category) => '$keyPrefix.$userId.${category.storageKey}';
     final SharedPreferencesAsync Function() _createPreferences;
     late final SharedPreferencesAsync _preferences = _createPreferences();
     @override
     Future<bool> isEnabled(String userId, NotificationCategory category) async =>
         await _preferences.getBool(keyFor(userId, category)) ?? defaultEnabled;
     @override
     Future<Map<NotificationCategory, bool>> readAll(String userId) async => {
       for (final category in NotificationCategory.values) category: await isEnabled(userId, category),
     };
     @override
     Future<void> setEnabled(String userId, NotificationCategory category, {required bool enabled}) =>
         _preferences.setBool(keyFor(userId, category), enabled);
   }
   ```

   ```dart
   /// Toggle notifiche del tab Impostazioni: ottimistici, salvati al tap (D-12).
   class NotificationPreferencesCubit extends Cubit<NotificationPreferencesState> with SafeEmit<NotificationPreferencesState> {
     NotificationPreferencesCubit(this._store, {required String userId}) : _userId = userId, super(const NotificationPreferencesState());
     final NotificationPreferencesStore _store;
     final String _userId;
     Future<void> load() async {
       if (state.isLoaded) return;
       try {
         final stored = await _store.readAll(_userId);
         emit(NotificationPreferencesState(enabled: Map.unmodifiable({...NotificationPreferencesState.defaults, ...stored}), isLoaded: true));
       } catch (error) {
         debugPrint('[NotificationPreferences] read failed: ${error.runtimeType}');
         emit(const NotificationPreferencesState(isLoaded: true));
       }
     }

     /// ravvicinati non si sovrascrivono. Nessuna UI d'errore (D-12).
     Future<void> toggle(NotificationCategory category, {required bool enabled}) async {
       if (!state.isLoaded || state.isEnabled(category) == enabled) return;
       emit(state.withCategory(category, enabled: enabled));
       try {
         await _store.setEnabled(_userId, category, enabled: enabled);
       } catch (error) {
         debugPrint('[NotificationPreferences] write failed: ${error.runtimeType}');
         emit(state.withCategory(category, enabled: !enabled));
       }
     }
   ```
